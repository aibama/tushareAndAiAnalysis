package com.example.utils;

import java.time.DayOfWeek;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.HashSet;
import java.util.Set;

/**
 * 国内交易所（A股）交易日判断工具类
 * 规则：周六周日休市 + 法定节假日休市
 */
public class ChinaStockTradingDayChecker {

    // 存放法定节假日 (格式：yyyy-MM-dd)
    private static final Set<String> HOLIDAYS = new HashSet<>();

    private static final DateTimeFormatter FORMATTER = DateTimeFormatter.ofPattern("yyyy-MM-dd");

    static {
        // 初始化2026年法定节假日 (根据交易所官方公告)
        // 元旦
        HOLIDAYS.add("2026-01-01");
        HOLIDAYS.add("2026-01-02");
        HOLIDAYS.add("2026-01-03");
        // 春节 (2月15日-2月23日)
        HOLIDAYS.add("2026-02-15");
        HOLIDAYS.add("2026-02-16");
        HOLIDAYS.add("2026-02-17");
        HOLIDAYS.add("2026-02-18");
        HOLIDAYS.add("2026-02-19");
        HOLIDAYS.add("2026-02-20");
        HOLIDAYS.add("2026-02-21");
        HOLIDAYS.add("2026-02-22");
        HOLIDAYS.add("2026-02-23");
        // 清明节
        HOLIDAYS.add("2026-04-04");
        HOLIDAYS.add("2026-04-05");
        HOLIDAYS.add("2026-04-06");
        // 劳动节
        HOLIDAYS.add("2026-05-01");
        HOLIDAYS.add("2026-05-02");
        HOLIDAYS.add("2026-05-03");
        HOLIDAYS.add("2026-05-04");
        HOLIDAYS.add("2026-05-05");
        // 端午节
        HOLIDAYS.add("2026-06-19");
        HOLIDAYS.add("2026-06-20");
        HOLIDAYS.add("2026-06-21");
        // 中秋节
        HOLIDAYS.add("2026-09-25");
        HOLIDAYS.add("2026-09-26");
        HOLIDAYS.add("2026-09-27");
        // 国庆节
        HOLIDAYS.add("2026-10-01");
        HOLIDAYS.add("2026-10-02");
        HOLIDAYS.add("2026-10-03");
        HOLIDAYS.add("2026-10-04");
        HOLIDAYS.add("2026-10-05");
        HOLIDAYS.add("2026-10-06");
        HOLIDAYS.add("2026-10-07");
    }

    /**
     * 判断指定日期是否为交易日
     * @param date 待判断日期
     * @return true: 是交易日, false: 不是交易日
     */
    public static boolean isTradingDay(LocalDate date) {
        // 1. 如果是周六或周日，直接返回false (非交易日)
        DayOfWeek dayOfWeek = date.getDayOfWeek();
        if (dayOfWeek == DayOfWeek.SATURDAY || dayOfWeek == DayOfWeek.SUNDAY) {
            return false;
        }

        // 2. 如果是周一至周五，检查是否在法定节假日列表中
        String dateStr = date.format(FORMATTER);
        return !HOLIDAYS.contains(dateStr);
    }

    /**
     * 判断指定日期字符串是否为交易日
     * @param dateStr 日期字符串 (格式：yyyy-MM-dd)
     * @return true: 是交易日, false: 不是交易日
     */
    public static boolean isTradingDay(String dateStr) {
        try {
            LocalDate date = LocalDate.parse(dateStr, FORMATTER);
            return isTradingDay(date);
        } catch (Exception e) {
            return false;
        }
    }

    /**
     * 获取从指定日期开始往前追溯的最近一个交易日
     * @param date 起始日期
     * @return 最近的一个交易日
     */
    public static LocalDate getPreviousTradingDay(LocalDate date) {
        LocalDate currentDate = date;
        while (!isTradingDay(currentDate)) {
            currentDate = currentDate.minusDays(1);
        }
        return currentDate;
    }

    /**
     * 获取从指定日期开始往前追溯的最近一个交易日
     * @param dateStr 起始日期字符串 (格式：yyyy-MM-dd)
     * @return 最近的一个交易日
     */
    public static LocalDate getPreviousTradingDay(String dateStr) {
        try {
            LocalDate date = LocalDate.parse(dateStr, FORMATTER);
            return getPreviousTradingDay(date);
        } catch (Exception e) {
            return LocalDate.now();
        }
    }

    /**
     * 获取从指定日期开始往后追溯的下一个交易日
     * @param date 起始日期
     * @return 下一个交易日
     */
    public static LocalDate getNextTradingDay(LocalDate date) {
        LocalDate currentDate = date.plusDays(1); // 从下一天开始找
        while (!isTradingDay(currentDate)) {
            currentDate = currentDate.plusDays(1);
        }
        return currentDate;
    }

    /**
     * 获取今天的日期字符串
     * @return 今天的日期 (格式：yyyy-MM-dd)
     */
    public static String getTodayDateStr() {
        return LocalDate.now().format(FORMATTER);
    }

    /**
     * 获取今天的LocalDate对象
     * @return 今天的LocalDate
     */
    public static LocalDate getToday() {
        return LocalDate.now();
    }

    /**
     * 获取最近一个交易日
     * 如果今天是交易日，返回今天；否则返回上一个交易日
     * @return 最近一个交易日的字符串（yyyy-MM-dd格式）
     */
    public static String getLatestTradingDayDateStr() {
        LocalDate today = getToday();
        if (isTradingDay(today)) {
            return today.format(FORMATTER);
        } else {
            LocalDate previousTradingDay = getPreviousTradingDay(today);
            return previousTradingDay.format(FORMATTER);
        }
    }

    /**
     * 判断是否需要更新数据
     * 根据Level2LastUpdateDate判断：
     * 1. 如果Level2LastUpdateDate等于最近一个交易日，无需更新
     * 2. 如果Level2LastUpdateDate为空或不是最近一个交易日，需要更新
     *
     * @param level2LastUpdateDate Level2最后更新时间
     * @return true: 需要更新数据, false: 无需更新
     */
    public static boolean needToUpdateData(String level2LastUpdateDate) {
        // 如果为空，需要更新
        if (level2LastUpdateDate == null || level2LastUpdateDate.trim().isEmpty()) {
            return true;
        }

        // 获取最近一个交易日
        String latestTradingDay = getLatestTradingDayDateStr();
        LocalDate lastUpdateDate;

        try {
            lastUpdateDate = LocalDate.parse(level2LastUpdateDate, FORMATTER);
        } catch (Exception e) {
            // 解析失败，需要更新
            return true;
        }

        // 如果等于最近一个交易日，无需更新
        if (lastUpdateDate.toString().equals(latestTradingDay)) {
            return false;
        }

        // 其他情况需要更新
        return true;
    }

    public static void main(String[] args) {
        // 测试几个关键日期
        LocalDate testDate1 = LocalDate.of(2026, 1, 1); // 元旦
        LocalDate testDate2 = LocalDate.of(2026, 2, 16); // 春节
        LocalDate testDate3 = LocalDate.of(2026, 5, 6); // 劳动节后第一天
        LocalDate testDate4 = LocalDate.of(2026, 9, 26); // 中秋节假期
        LocalDate testDate5 = LocalDate.of(2026, 10, 9); // 国庆节后

        System.out.println(testDate1 + " 是否为交易日？ " + isTradingDay(testDate1));
        System.out.println(testDate2 + " 是否为交易日？ " + isTradingDay(testDate2));
        System.out.println(testDate3 + " 是否为交易日？ " + isTradingDay(testDate3));
        System.out.println(testDate4 + " 是否为交易日？ " + isTradingDay(testDate4));
        System.out.println(testDate5 + " 是否为交易日？ " + isTradingDay(testDate5));

        System.out.println("\n从 " + testDate1 + " 往前找最近交易日：" + getPreviousTradingDay(testDate1));
        System.out.println("从 " + testDate4 + " 往后找下一个交易日：" + getNextTradingDay(testDate4));

        // 测试needToUpdateData方法
        System.out.println("\n========== 测试needToUpdateData ==========");
        System.out.println("今天(" + getTodayDateStr() + "), lastUpdate为空: " + needToUpdateData(""));
        System.out.println("今天(" + getTodayDateStr() + "), lastUpdate为今天: " + needToUpdateData(getTodayDateStr()));
        System.out.println("今天(" + getTodayDateStr() + "), lastUpdate为昨天: " + needToUpdateData(LocalDate.now().minusDays(1).format(FORMATTER)));
    }
}
